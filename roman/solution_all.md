# 🧩 ПОДРОБНЫЙ РАЗБОР КОДА

```
#include <iostream>
#include <string>
#include <unordered_map>
using namespace std;

// Перевод римской строки в число (без проверки валидности)
int romanToInt(const string& s) {
    unordered_map<char, int> value = {
        {'I', 1}, {'V', 5}, {'X', 10},
        {'L', 50}, {'C', 100}, {'D', 500}, {'M', 1000}
    };
    int ans = 0;
    for (int i = 0; i < (int)s.size(); ++i) {
        if (i + 1 < (int)s.size() && value[s[i]] < value[s[i + 1]]) {
            ans -= value[s[i]];
        } else {
            ans += value[s[i]];
        }
    }
    return ans;
}

// Перевод числа в каноническую римскую запись (только для чисел < 4000)
string intToRoman(int num) {
    pair<int, string> values[] = {
        {1000, "M"}, {900, "CM"}, {500, "D"}, {400, "CD"},
        {100, "C"}, {90, "XC"}, {50, "L"}, {40, "XL"},
        {10, "X"}, {9, "IX"}, {5, "V"}, {4, "IV"}, {1, "I"}
    };
    string roman;
    for (auto &p : values) {
        while (num >= p.first) {
            roman += p.second;
            num -= p.first;
        }
    }
    return roman;
}

int convertToArabic(const string& romanNumber) {
    // Проверка на недопустимые символы
    unordered_map<char, int> value = {
        {'I', 1}, {'V', 5}, {'X', 10},
        {'L', 50}, {'C', 100}, {'D', 500}, {'M', 1000}
    };
    for (char c : romanNumber) {
        if (value.find(c) == value.end()) {
            return -1;
        }
    }

    int num = romanToInt(romanNumber);

    // Числа ≥ 4000 не могут быть записаны по правилам задачи
    if (num >= 4000) {
        return -1;
    }

    string canonical = intToRoman(num);

    if (canonical == romanNumber) {
        return num;
    }
    return -1;
}

int main() {
    string romanNumber;
    cin >> romanNumber;
    cout << convertToArabic(romanNumber);
}
```

## ЧАСТЬ 2: ФУНКЦИЯ romanToInt (ПЕРЕВОД РИМСКОЙ СТРОКИ В ЧИСЛО)

```cpp
int romanToInt(const string& s) {
```

**Что здесь происходит:**
- `int` — функция возвращает целое число
- `romanToInt` — имя функции (переводит римскую строку в число)
- `const string& s` — принимает строку по ссылке (чтобы не копировать)
- `const` — обещает не изменять строку

---

```cpp
    unordered_map<char, int> value = {
        {'I', 1}, {'V', 5}, {'X', 10},
        {'L', 50}, {'C', 100}, {'D', 500}, {'M', 1000}
    };
```

**Что здесь происходит:**
- `unordered_map<char, int>` — создаём хеш-таблицу, где ключ — символ (`char`), значение — число (`int`)
- `value` — имя нашей таблицы
- `{ {'I', 1}, ... }` — заполняем таблицу парами "символ → число"

**Визуализация:**
```
value = {
    'I' → 1,
    'V' → 5,
    'X' → 10,
    'L' → 50,
    'C' → 100,
    'D' → 500,
    'M' → 1000
}
```

**Зачем это нужно:** Мы хотим быстро узнать, сколько стоит каждая римская цифра. Например, `value['I']` вернёт 1.

---

```cpp
    int ans = 0;
```

**Что здесь происходит:**
- `int ans` — создаём переменную для ответа (суммы)
- `= 0` — начинаем с нуля

**Пример:** `ans` будет накапливать результат.

---

```cpp
    for (int i = 0; i < (int)s.size(); ++i) {
```

**Что здесь происходит:**
- `int i = 0` — создаём счётчик, начинаем с 0
- `i < (int)s.size()` — пока не дошли до конца строки
- `++i` — увеличиваем счётчик на 1 после каждой итерации

**Что такое `(int)s.size()`:**
- `s.size()` — возвращает длину строки (тип `size_t`)
- `(int)` — преобразуем в обычное целое число, чтобы избежать предупреждений

---

```cpp
        if (i + 1 < (int)s.size() && value[s[i]] < value[s[i + 1]]) {
            ans -= value[s[i]];
        } else {
            ans += value[s[i]];
        }
    }
```

**Разберём по частям:**

### Шаг 1: Проверка условия
```cpp
if (i + 1 < (int)s.size() && value[s[i]] < value[s[i + 1]]) {
```
- `i + 1 < (int)s.size()` — проверяем, есть ли **следующая** цифра
- `value[s[i]] < value[s[i + 1]]` — сравниваем значения: текущая цифра **меньше** следующей?

**Если оба условия верны:**
```cpp
ans -= value[s[i]];  // вычитаем текущую цифру
```
**Пример:** `IV`
- `i = 0`, `s[0] = 'I'`, `s[1] = 'V'`
- `value['I'] = 1`, `value['V'] = 5`
- `1 < 5` → правда → `ans -= 1` → `ans = -1`
- Затем на следующей итерации прибавим `V` = 5 → итого 4

**Если условие ложно:**
```cpp
ans += value[s[i]];  // прибавляем текущую цифру
```
**Пример:** `VI`
- `i = 0`, `s[0] = 'V'`, `s[1] = 'I'`
- `value['V'] = 5`, `value['I'] = 1`
- `5 < 1` → ложь → `ans += 5` → `ans = 5`
- `i = 1` → `ans += 1` → `ans = 6`

---

```cpp
    return ans;
}
```

**Что здесь происходит:**
- Возвращаем накопленную сумму

**Пример:**
- Вход: `"XIV"` (14)
- Итерации:
  - `X` (10) → следующая `I` (1) → `10 < 1`? Нет → `ans = 10`
  - `I` (1) → следующая `V` (5) → `1 < 5`? Да → `ans = 10 - 1 = 9`
  - `V` (5) → нет следующей → `ans = 9 + 5 = 14`
- Возвращает 14 ✅

## ЧАСТЬ 3: ФУНКЦИЯ intToRoman (ПЕРЕВОД ЧИСЛА В РИМСКУЮ СТРОКУ)

```cpp
string intToRoman(int num) {
```

**Что здесь происходит:**
- `string` — функция возвращает строку
- `intToRoman` — имя функции
- `int num` — принимает число для перевода

---

```cpp
    pair<int, string> values[] = {
        {1000, "M"}, {900, "CM"}, {500, "D"}, {400, "CD"},
        {100, "C"}, {90, "XC"}, {50, "L"}, {40, "XL"},
        {10, "X"}, {9, "IX"}, {5, "V"}, {4, "IV"}, {1, "I"}
    };
```

**Что здесь происходит:**
- `pair<int, string>` — пара: число и соответствующая римская строка
- `values[]` — массив таких пар

**Визуализация:**
```
values = [
    (1000, "M"),
    (900, "CM"),
    (500, "D"),
    (400, "CD"),
    ...
]
```

**Зачем этот порядок?** От **больших** чисел к **меньшим** (жадный алгоритм). Это гарантирует правильную запись.

---

```cpp
    string roman;
```

**Что здесь происходит:**
- `string roman` — создаём пустую строку для результата

---

```cpp
    for (auto &p : values) {
        while (num >= p.first) {
            roman += p.second;
            num -= p.first;
        }
    }
```

**Разберём по частям:**

### Внешний цикл:
```cpp
for (auto &p : values) {
```
- `auto &p` — переменная `p` проходит по всем парам в `values`
- `&` — ссылка (чтобы не копировать)

### Внутренний цикл:
```cpp
while (num >= p.first) {
    roman += p.second;
    num -= p.first;
}
```
- `while (num >= p.first)` — пока число больше или равно текущему значению
- `roman += p.second` — добавляем римскую строку к ответу
- `num -= p.first` — вычитаем значение из числа

**Пример:** `num = 1994`

| Шаг | `num` | Пара | Условие | Действие | `roman` | `num` после |
|-----|-------|------|---------|----------|---------|-------------|
| 1 | 1994 | (1000, "M") | 1994 >= 1000 | +"M", -1000 | `"M"` | 994 |
| 2 | 994 | (900, "CM") | 994 >= 900 | +"CM", -900 | `"MCM"` | 94 |
| 3 | 94 | (90, "XC") | 94 >= 90 | +"XC", -90 | `"MCMXC"` | 4 |
| 4 | 4 | (4, "IV") | 4 >= 4 | +"IV", -4 | `"MCMXCIV"` | 0 |

**Результат:** `"MCMXCIV"` (1994) ✅

---

```cpp
    return roman;
}
```

**Что здесь происходит:**
- Возвращаем собранную строку

## ЧАСТЬ 4: ФУНКЦИЯ convertToArabic (ГЛАВНАЯ)

```cpp
int convertToArabic(const string& romanNumber) {
```

**Что здесь происходит:**
- `int` — возвращает число или `-1` (если невалидно)
- `convertToArabic` — имя функции
- `const string& romanNumber` — принимает римскую строку по ссылке

---

```cpp
    unordered_map<char, int> value = {
        {'I', 1}, {'V', 5}, {'X', 10},
        {'L', 50}, {'C', 100}, {'D', 500}, {'M', 1000}
    };
```

**Что здесь происходит:**
- Создаём такую же таблицу значений (как в `romanToInt`)

---

```cpp
    for (char c : romanNumber) {
        if (value.find(c) == value.end()) {
            return -1;
        }
    }
```

**Что здесь происходит:**
- `for (char c : romanNumber)` — проходим по каждому символу `c` в строке `romanNumber`
- `value.find(c)` — ищем символ `c` в таблице `value`
- `.end()` — если не нашли (дошли до конца таблицы)
- `return -1` — символ недопустимый → возвращаем `-1`

**Пример:**
- `romanNumber = "IVX"` → все символы есть → продолжаем
- `romanNumber = "IVA"` → `'A'` нет в таблице → возвращаем `-1`

---

```cpp
    int num = romanToInt(romanNumber);
```

**Что здесь происходит:**
- `romanToInt(romanNumber)` — переводим строку в число
- `num` — сохраняем результат

**Пример:**
- `romanNumber = "IV"` → `num = 4`

---

```cpp
    if (num >= 4000) {
        return -1;
    }
```

**Что здесь происходит:**
- Проверяем, не больше ли число 4000
- По условию задачи, римская запись не может представлять числа ≥ 4000 (нужно 4 `M` подряд, а нельзя)
- Если `num >= 4000` → возвращаем `-1`

**Пример:**
- `romanNumber = "MMMM"` → `num = 4000` → возвращаем `-1` ✅

---

```cpp
    string canonical = intToRoman(num);
```

**Что здесь происходит:**
- `intToRoman(num)` — строим **каноническую** (правильную) римскую запись числа
- `canonical` — сохраняем результат

**Пример:**
- `num = 4` → `intToRoman(4)` = `"IV"`

---

```cpp
    if (canonical == romanNumber) {
        return num;
    }
    return -1;
}
```

**Что здесь происходит:**
- `canonical == romanNumber` — сравниваем каноническую запись с исходной строкой
- Если **совпадают** → строка валидна → возвращаем число
- Если **не совпадают** → строка невалидна → возвращаем `-1`

**Примеры:**
| Вход | `romanToInt` | `intToRoman` | Сравнение | Результат |
|------|--------------|--------------|-----------|-----------|
| `"IV"` | 4 | `"IV"` | ✅ | 4 |
| `"IVI"` | 5 | `"V"` | ❌ | -1 |
| `"MCMXCIV"` | 1994 | `"MCMXCIV"` | ✅ | 1994 |
